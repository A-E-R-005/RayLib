You need a separate `Texture2D` and a separate `SheetAnim` object for each state (`IDLE` and `WALK`).

`main.cpp` (Animation Instantiation and Logic)
	In `main.cpp`, you must:
		1. Load the Walk Texture and create the `playerA_walk` animation instance
		2. Use the `if/else` logic in the main loop to run the correct `Alogic()` and drawing based on `player.currentState`

```c++
// main.cpp

int main(void)
{
    // I. INITIALIZATION
    InitWindow(800,800,"Test 8");
    SetTargetFPS(60);

    Player player;
    player.loading(); 
    player.currentState = IDLE; // Initialize state
    
    // Target setup remains the same
    Target target;
    target.loading();
    SheetAnim targetA;
    targetA.Aloading(6,&target.targetTexture);
    
    // 1. Create separate animation objects for each state
    SheetAnim playerA_idle;
    playerA_idle.Aloading(6, &player.playerTexture); // Uses Soldier-Idle.png (loaded in player.loading)
    
    // 2. Load the Walk Texture separately
    Texture2D walkTexture = LoadTexture("../assets/Characters/Soldier/Soldier/Soldier-Walk.png"); 
    SheetAnim playerA_walk;
    playerA_walk.Aloading(6, &walkTexture); 
    
    while (!WindowShouldClose())
    {
        // ... (Collision and Boundary checks) ...
        player.Movement(); // <<< This updates player.currentState

        // 3. ANIMATION: Only run logic for the current state
        if (player.currentState == IDLE) {
            playerA_idle.Alogic(); // Runs idle timer/frame advance
        } else {
            playerA_walk.Alogic(); // Runs walk timer/frame advance
        }
        
        targetA.Alogic(); 

        // DRAWING
        BeginDrawing();
        ClearBackground(RAYWHITE);
        target.Draw(targetA.sourceRect);

        // 4. Draw: Select the active animation source and texture
        SheetAnim* activeAnim = (player.currentState == IDLE) ? &playerA_idle : &playerA_walk;
        
        // The Draw function must use the texture and sourceRect from the active animation!
        // NOTE: Since your Player::Draw only takes sourceRect, you need to update it
        //       to accept the correct Texture2D as well, or assign it before drawing.
        
        // TEMPORARY FIX: Assign the correct texture before drawing
        player.playerTexture = activeAnim->spriteSheet[0]; // <<< This assumes SheetAnim holds a pointer to the texture. Needs cleanup.
        
        // Correct Draw Call (Passing all necessary info):
        player.Draw(activeAnim->sourceRect, activeAnim->spriteSheet[0]); 
        
        EndDrawing();
    }
    // DE-INITIALIZATION
    UnloadTexture(player.playerTexture); // Unloads Idle texture
    UnloadTexture(walkTexture);          // Unload Walk texture
    UnloadTexture(target.targetTexture);

    CloseWindow();
    return 0;
}
```
